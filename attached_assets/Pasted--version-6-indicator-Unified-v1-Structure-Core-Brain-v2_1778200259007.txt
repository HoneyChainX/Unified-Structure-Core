f_mul_tf(_k) =>
    int mult = timeframe.multiplier

    if timeframe.isseconds
        int s = mult * _k
        s % 86400 == 0 ? str.tostring(s / 86400) + "D" :
         s % 3600  == 0 ? str.tostring(s / 3600)  + "H" :
         s % 60    == 0 ? str.tostring(s / 60)          :
                          str.tostring(s) + "S"

    else if timeframe.isminutes
        int m = mult * _k
        m % 10080 == 0 ? str.tostring(m / 10080) + "W" :
         m % 1440  == 0 ? str.tostring(m / 1440)  + "D" :
                          str.tostring(m)

    else if timeframe.isdaily
        str.tostring(mult * _k) + "D"
    else if timeframe.isweekly
        str.tostring(mult * _k) + "W"
    else if timeframe.ismonthly
        str.tostring(mult * _k) + "M"
    else
        str.tostring(mult * _k)

f_tfMinutes() =>
    int out = timeframe.multiplier
    if timeframe.isseconds
        out := math.max(1, int(timeframe.multiplier / 60))
    else if timeframe.isminutes
        out := timeframe.multiplier
    else if timeframe.isdaily
        out := timeframe.multiplier * 1440
    else if timeframe.isweekly
        out := timeframe.multiplier * 10080
    else if timeframe.ismonthly
        out := timeframe.multiplier * 43200 // 30d * 1440
    else
        out := timeframe.multiplier
    out

f_mss(_len) =>
    [ph, pl] = f_pivots(_len)

    var float lastHigh = na
    var float lastLow  = na
    var int   dir      = 0
    var float mssPrice = na
    var int   mssBar   = na

    if not na(ph)
        lastHigh := ph
    if not na(pl)
        lastLow := pl

    bool bullBreak = not na(lastHigh) and close > lastHigh
    bool bearBreak = not na(lastLow)  and close < lastLow

    if bullBreak
        dir := 1
        mssPrice := lastHigh
        mssBar := bar_index
    else if bearBreak
        dir := -1
        mssPrice := lastLow
        mssBar := bar_index

    [dir, mssPrice, mssBar]

f_find_ob(_dir, _lookback) =>
    float obLow  = na
    float obHigh = na
    int   obBar  = na

    if _dir == 1
        for i = 1 to _lookback
            if close[i] < open[i]
                obLow  := math.min(open[i], close[i])
                obHigh := math.max(open[i], close[i])
                obBar  := bar_index - i
                break
    else if _dir == -1
        for i = 1 to _lookback
            if close[i] > open[i]
                obLow  := math.min(open[i], close[i])
                obHigh := math.max(open[i], close[i])
                obBar  := bar_index - i
                break

    [obLow, obHigh, obBar]

float zLo = ENTRY_ZONE_LOW  - marginAbs
float zHi = ENTRY_ZONE_HIGH + marginAbs

bool  zoneOK = not na(ENTRY_ZONE_LOW) and not na(ENTRY_ZONE_HIGH) and (priceX >= zLo) and (priceX <= zHi)

// Regular divergences (synced)
bool bullDiv_reg = syncLow  and not na(pLo1) and not na(pLo2) and not na(sLo1) and not na(sLo2) and (pLo1 < pLo2) and (sLo1 > sLo2)
bool bearDiv_reg = syncHigh and not na(pHi1) and not na(pHi2) and not na(sHi1) and not na(sHi2) and (pHi1 > pHi2) and (sHi1 < sHi2)

// Hidden divergences (synced) — trend continuation
// Hidden Bull: price HL, score LL
bool bullDiv_hid = syncLow  and not na(pLo1) and not na(pLo2) and not na(sLo1) and not na(sLo2) and (pLo1 > pLo2) and (sLo1 < sLo2)
// Hidden Bear: price LH, score HH
bool bearDiv_hid = syncHigh and not na(pHi1) and not na(pHi2) and not na(sHi1) and not na(sHi2) and (pHi1 < pHi2) and (sHi1 > sHi2)

// Direction gating + zone gating
bool bullOK = (PA_DIR == 1) and zoneOK
bool bearOK = (PA_DIR == -1) and zoneOK

bool bullDiv = bullOK and (bullDiv_reg or (useHiddenDiv and bullDiv_hid))
bool bearDiv = bearOK and (bearDiv_reg or (useHiddenDiv and bearDiv_hid))

bool divSupports =
     not useDivergence ? true :
     (PA_DIR == 1 ? bullDiv : PA_DIR == -1 ? bearDiv : false)

// Confidence / Confluence
int CONF_SCORE = 0
CONF_SCORE += (PA_DIR != 0) ? 1 : 0
CONF_SCORE += biasAligned ? 1 : 0

bool ctxDirOK = (PA_DIR == 1 ? (CTX != "Hostile") : PA_DIR == -1 ? (CTX != "Supportive") : false)
CONF_SCORE += ctxDirOK ? 1 : 0
CONF_SCORE += IN_ZONE ? 1 : 0

bool ema1OK = (PA_DIR == 1 ? emaTF1Long : PA_DIR == -1 ? emaTF1Short : false)
CONF_SCORE += ema1OK ? 1 : 0

bool ema3OK = (PA_DIR == 1 ? emaTF3Long : PA_DIR == -1 ? emaTF3Short : false)
CONF_SCORE += ema3OK ? 1 : 0

CONF_SCORE += (useDivergence and divSupports) ? 1 : 0

string SETUP_GRADE_RAW =
     CONF_SCORE >= 6 ? "A+ Setup" :
     CONF_SCORE >= 4 ? "Strong Setup" :
     CONF_SCORE >= 3 ? "Setup" :
     "None"

bool SIGNAL_OK = (SETUP_GRADE_RAW == "Strong Setup") or (SETUP_GRADE_RAW == "A+ Setup")

// Weighted Confidence (0..100)
float wb = (PA_DIR != 0 and biasAligned) ? 1.0 : 0.0
float wc = ctxDirOK ? 1.0 : 0.0
float wz = IN_ZONE ? 1.0 : 0.0
float we = (ema1OK ? 0.5 : 0.0) + (ema3OK ? 0.5 : 0.0)          // EMA total 0..1
float wd = (useDivergence and divSupports) ? 1.0 : 0.0

// Rotation support (only meaningful when rotation layer enabled)
bool rotSupport =
     not useRotationLayer ? true :
     (PA_DIR == 1 ? (ROT_STATE != "RISK_OFF") : true)

float wr = rotSupport ? 1.0 : 0.0

float W_SUM = (w_bias + w_ctx + w_zone + w_ema + w_div + w_rot)
float W_HIT = (wb*w_bias) + (wc*w_ctx) + (wz*w_zone) + (we*w_ema) + (wd*w_div) + (wr*w_rot)

// Batch-1: W_SUM guard (hardening only)
float CONF_W = na
if useWeightedConfidence
    CONF_W := (W_SUM != 0) ? (100.0 * W_HIT / W_SUM) : 0.0
else
    CONF_W := (100.0 * CONF_SCORE / 7.0)

string RISK_TIER =
     CONF_W >= confHiTh ? "HIGH_CONF" :
     CONF_W >= confMdTh ? "MID_CONF" :
     "LOW_CONF"

//────────────────────────────────────────────────────────────────────────────
// Gates v2 (B > C > Z): sequential evaluation + explicit reason
//────────────────────────────────────────────────────────────────────────────
bool GATE_BIAS = biasAligned
bool GATE_CTX  = GATE_BIAS and ctxDirOK
bool GATE_ZONE = GATE_CTX and IN_ZONE

bool TRIGGER_OK = SIGNAL_OK and ENTRY_ALLOWED and GATE_ZONE

TRIGGER_OK := TRIGGER_OK and ROT_GATE_OK

     not IN_ZONE        ? "Z_FAIL" :
     not SIGNAL_OK      ? "CONF_FAIL" :
     not ROT_GATE_OK    ? "ROT_FAIL" :
     "OK"
